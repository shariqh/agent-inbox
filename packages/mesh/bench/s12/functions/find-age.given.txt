// An age in the row's narrow column: now, 5m, 3h, 2d, 3w.
export function findAge(ageMs) {
  // Convert the age from milliseconds to minutes
  const m = Math.floor(Math.max(0, ageMs) / 60000)
  // Check if the age is less than a minute
  if (m < 1) return 'now'
  if (m < 60) return `${m}m`
  // Calculate the hours
  const h = Math.floor(m / 60)
  if (h < 24) return `${h}h`
  const d = Math.floor(h / 24)
  if (d < 7) return `${d}d`
  // Finally, return the age in weeks
  return `${Math.floor(d / 7)}w`
}
